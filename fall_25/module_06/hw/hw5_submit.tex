\subsubsection*{Solution 1 (c)}
We can solve this using the Empirical Rule (68-95-99.7 rule).
Given $X \sim N(10, 16)$, we have $\mu = 10$ and $\sigma = \sqrt{16} = 4$.
We want to find $P(X \geq 14)$, which is $P(X \geq \mu + \sigma)$.


\begin{flushleft}
    The Empirical Rule states that approximately 68\% of the data lies within one standard deviation of the mean.
\end{flushleft}
\begin{align*}
    P(\mu - \sigma \leq X \leq \mu + \sigma) &\approx 0.68 \\
    P(6 \leq X \leq 14) &\approx 0.68
\end{align*}
The total probability outside this range is $1 - 0.68 = 0.32$.
\begin{align*}
    P(X < 6) + P(X > 14) \approx 0.32
\end{align*}
By the symmetry of the Gaussian distribution, the two tail probabilities are equal.
\begin{align*}
    P(X \geq 14) = P(X > 14) \approx \frac{0.32}{2} = 0.16
\end{align*}

\subsubsection*{\normalfont}{$\therefore$ $P(X \geq 14) \approx 0.16$}

\subsubsection*{Solution 3 (a)}
\begin{flushleft}
    For any probability density function (PDF), the total area under the curve must integrate to 1.
\end{flushleft}

$$\int_{-\infty}^{\infty} f(x) \,dx = 1$$

\begin{flushleft}
    We are given the bounds of the probability density function $[0, \lambda]$, so the equation becomes:
\end{flushleft}

\begin{align*}
    \int_{0}^{\lambda} c \,dx &= 1 \\
    c \cdot [x]_{0}^{\lambda} &= 1 \\
    c \cdot (\lambda - 0) &= 1 \\
    c \cdot \lambda &= 1 \\
    c &= \frac{1}{\lambda}
\end{align*}

\subsubsection*{\normalfont}{$\therefore$ the density of $U_{\lambda}$ is $\frac{1}{\lambda}$ and we can express it as $f(x \mid \lambda)$:}

$$
f(x \mid \lambda) = 
\begin{cases} 
      \frac{1}{\lambda} & \text{if } 0 \leq x \leq \lambda \\
      0 & \text{otherwise} 
\end{cases}
$$

\subsubsection*{Solution 3 (b)}

From \textit{part (a) } $U_{\lambda}$ on $[0,\lambda]$, the density is
$$
f(x \mid \lambda) = 
\begin{cases} 
      \frac{1}{\lambda} & \text{if } 0 \leq x \leq \lambda \\
      0 & \text{otherwise} 
\end{cases}
$$
The likelihood function $L(\lambda)$ for a set of $n$ independent observations $X = \{x_1, \ldots, x_n\}$ is the product of the individual densities:
\begin{align*}
    L(\lambda \mid X) = \prod_{i=1}^n f(x_i \mid \lambda)
\end{align*}
\begin{flushleft}
    For this likelihood to be non-zero, every single term $f(x_i \mid \lambda)$ must be non-zero. Based on the PDF definition, this means we must have $0 \leq x_i \leq \lambda$ for all $i = 1, \ldots, n$.
\end{flushleft}

\begin{flushleft}
    If we choose a $\lambda$ that is smaller than even one observation, say $x_j$, then $f(x_j \mid \lambda) = 0$, and the entire likelihood $L(\lambda) = 0$. This is the minimum possible likelihood, not the maximum.
\end{flushleft}

\begin{flushleft}
    It follows that, to have a non-zero likelihood, the choice of $\lambda$ must be greater than or equal to all observations or:
\end{flushleft}

$$
\lambda \geq \max(x_1, \ldots, x_n)
$$

\subsubsection*{\normalfont}{$\therefore$ the max likelihood choice of ${\lambda}$ is $\max(x_1, x_2, \ldots, x_n)$}

\subsubsection*{Question 8 (b)}
\begin{flushleft}
Let $\theta$ denote the probability that you win when playing tennis with your friend. You want to estimate this probability, and you decide to try both frequentist and Bayesian approaches.

You play ten matches with your friend. The outcomes are:
\end{flushleft}
$$X = (W, W, L, L, L, L, L, W, W, W)$$

\begin{flushleft}
where $W$ means that you win and $L$ means that you lose.\\

Now let's follow a Bayesian approach. This time, we need to begin with a prior distribution on the quantity of interest, $\theta$. Based on your prior knowledge about your relative tennis abilities, you put a $\operatorname{Beta}(6,4)$ prior distribution on $\theta$.

What is $\mathbb{E}[\theta]$, the mean of the prior distribution? What is its standard deviation, $\operatorname{std}(\theta)$?

\end{flushleft}

\subsubsection*{Solution 8 (b)}
Given the prior distribution $\theta \sim \operatorname{Beta}(\alpha, \beta)$, with $\alpha = 6$ and $\beta = 4$.

\subsubsection*{Mean}
The mean of a Beta distribution is given by the formula:
\begin{align*}
    \mathbb{E}[\theta] = \frac{\alpha}{\alpha + \beta}
\end{align*}
Substituting our values:
\begin{align*}
    \mathbb{E}[\theta] = \frac{6}{6 + 4} = \frac{6}{10} = 0.6
\end{align*}
This prior suggests a belief that you are more likely to win than to lose (mean = 0.6).

\subsubsection*{Standard Deviation}
The variance of a Beta distribution is given by:
\begin{align*}
    \operatorname{Var}(\theta) = \frac{\alpha \beta}{(\alpha + \beta)^2 (\alpha + \beta + 1)}
\end{align*}
Substituting our values:
\begin{align*}
    \operatorname{Var}(\theta) &= \frac{6 \times 4}{(6 + 4)^2 (6 + 4 + 1)} \\
                         &= \frac{24}{(10)^2 (11)} \\
                         &= \frac{24}{100 \times 11} = \frac{24}{1100} \approx 0.0218
\end{align*}
The standard deviation is the square root of the variance:
\begin{align*}
    \operatorname{std}(\theta) = \sqrt{\operatorname{Var}(\theta)} = \sqrt{\frac{24}{1100}} \approx 0.1477
\end{align*}

\subsubsection*{\normalfont}{$\therefore$ The prior mean is $\mathbb{E}[\theta] = 0.6$ and the prior standard deviation is $\operatorname{std}(\theta) = \sqrt{\frac{24}{1100}} \approx 0.1477$.} \\

\subsubsection*{Solution 8}
\noindent\rule{\textwidth}{0.4pt}\\
\subsubsection*{Solution 8 (c)}
This problem uses the Beta-Binomial conjugate model.

\begin{itemize}
    \item \textbf{Prior:} The prior distribution for $\theta$ is $P(\theta) \sim \operatorname{Beta}(\alpha, \beta)$, with $\alpha = 6$ and $\beta = 4$.
    \item \textbf{Likelihood:} The data $X$ consists of $n_W = 5$ wins and $n_L = 5$ losses. The likelihood of this data given $\theta$ is proportional to a Binomial, $P(X \mid \theta) \propto \theta^{n_W} (1-\theta)^{n_L} = \theta^5 (1-\theta)^5$.
    \item \textbf{Posterior:} The posterior $P(\theta \mid X)$ is proportional to the prior times the likelihood.
    \begin{align*}
        P(\theta \mid X) &\propto P(X \mid \theta) \times P(\theta) \\
                       &\propto [\theta^{n_W} (1-\theta)^{n_L}] \times [\theta^{\alpha-1} (1-\theta)^{\beta-1}] \\
                       &\propto \theta^{\alpha + n_W - 1} (1-\theta)^{\beta + n_L - 1}
    \end{align*}
    This has the form of a Beta distribution with updated parameters $\alpha' = \alpha + n_W$ and $\beta' = \beta + n_L$.
\end{itemize}

\subsubsection*{posterior distribution}
We update the parameters with our prior ($\alpha=6, \beta=4$) and data ($n_W=5, n_L=5$):
\begin{align*}
    \alpha' &= \alpha + n_W = 6 + 5 = 11 \\
    \beta' &= \beta + n_L = 4 + 5 = 9
\end{align*}
The posterior distribution is $\operatorname{Beta}(11, 9)$.

\subsubsection*{Mean of posterior distribution}
We use the mean formula for a Beta distribution with our posterior parameters $\alpha' = 11$ and $\beta' = 9$:
\begin{align*}
    \mathbb{E}[\theta \mid X] = \frac{\alpha'}{\alpha' + \beta'} = \frac{11}{11 + 9} = \frac{11}{20} = 0.55
\end{align*}

\subsubsection*{Standard Deviation posterior distribution}
We use the variance formula with posterior parameters:
\begin{align*}
    \operatorname{Var}(\theta \mid X) &= \frac{\alpha' \beta'}{(\alpha' + \beta')^2 (\alpha' + \beta' + 1)} \\
                               &= \frac{11 \times 9}{(11 + 9)^2 (11 + 9 + 1)} \\
                               &= \frac{99}{(20)^2 (21)} \\
                               &= \frac{99}{400 \times 21} = \frac{99}{8400}
\end{align*}
The standard deviation is the square root of the variance:
\begin{align*}
    \operatorname{std}(\theta \mid X) = \sqrt{\frac{99}{8400}} \approx 0.1085
\end{align*}
Notice that the posterior standard deviation (0.1085) is smaller than the prior standard deviation (0.1477), which shows that observing data has reduced our uncertainty about $\theta$.

\subsubsection*{\normalfont}{$\therefore$ The posterior distribution is $\operatorname{Beta}(11, 9)$, the posterior mean is $0.55$, and the posterior standard deviation is $\sqrt{\frac{99}{8400}} \approx 0.1085$.} \\

\begin{align*}
    \alpha' &= \alpha + \sum x_i = 1 + 94 = 95 \\
    \beta' &= \beta + n = 1 + 10 = 11

\subsubsection*{Question 10 (a)}
\begin{flushleft}
In an election, there are three candidates, $A, B$, and $C$. Of the voting population, $\theta_{A}$ fraction will vote for $A$, while $\theta_{B}$ fraction will vote for $B$ and $\theta_{C}$ fraction will vote for $C$; thus $\theta_{A}+\theta_{B}+\theta_{C}=1$.\\

John wishes to infer the distribution $\left(\theta_{A}, \theta_{B}, \theta_{C}\right)$. He starts by putting a uniform prior on it; that is, he considers all distributions to be equally likely. He then picks four people at random to poll, and finds that they intend to vote for $A, B, B, A$, respectively. Given this information, what is John's posterior distribution?
\end{flushleft}

\subsubsection*{Solution 10 (a)}
This is a Bayesian inference problem for a Multinomial distribution. The vector $\vec{\theta} = (\theta_A, \theta_B, \theta_C)$ represents the parameters of the multinomial likelihood.

\begin{itemize}
    \item \textbf{Likelihood:} The likelihood of the data $X = (A, B, B, A)$ given the parameters $\vec{\theta}$ is $P(X \mid \vec{\theta}) \propto \theta_A^{n_A} \theta_B^{n_B} \theta_C^{n_C}$, where $n_i$ is the count for candidate $i$.
    \item \textbf{Prior:} The conjugate prior for a Multinomial likelihood is the Dirichlet distribution, $P(\vec{\theta}) \sim \operatorname{Dirichlet}(\vec{\alpha})$. A "uniform prior" over the simplex (where $\sum \theta_i = 1$) is a special case of the Dirichlet distribution where all concentration parameters are 1.
    \begin{align*}
        P(\vec{\theta}) \sim \operatorname{Dirichlet}(\alpha_A, \alpha_B, \alpha_C) = \operatorname{Dirichlet}(1, 1, 1)
    \end{align*}
    \item \textbf{Data:} We observe the data $X = (A, B, B, A)$. We extract the counts:
    \begin{itemize}
        \item $n_A = 2$
        \item $n_B = 2$
        \item $n_C = 0$
    \end{itemize}
    \item \textbf{Posterior:} The posterior distribution is also a Dirichlet distribution. We update the parameters by adding the observed counts to the prior parameters.
    \begin{align*}
        P(\vec{\theta} \mid X) &\sim \operatorname{Dirichlet}(\alpha_A + n_A, \alpha_B + n_B, \alpha_C + n_C) \\
                              &\sim \operatorname{Dirichlet}(1 + 2, 1 + 2, 1 + 0) \\
                              &\sim \operatorname{Dirichlet}(3, 3, 1)
    \end{align*}
\end{itemize}
This result is equivalent to performing Maximum Likelihood with Laplace (add-one) smoothing, which is a common property when using a uniform Dirichlet prior.

\subsubsection*{\normalfont}{$\therefore$ John's posterior distribution is $\operatorname{Dirichlet}(3, 3, 1)$.} \\

Under this posterior distribution, what is the expected value of $\left(\theta_{A}, \theta_{B}, \theta_{C}\right)$?

From part (a), our posterior distribution is $\vec{\theta} \mid X \sim \operatorname{Dirichlet}(\vec{\alpha}')$, where $\vec{\alpha}' = (\alpha'_A, \alpha'_B, \alpha'_C) = (3, 3, 1)$.

The formula for the expected value of a component $\theta_i$ from a Dirichlet distribution is:
\begin{align*}
    \mathbb{E}[\theta_i \mid X] = \frac{\alpha'_i}{\sum_{j} \alpha'_j}
\end{align*}
First, we compute the sum of the posterior parameters, $\alpha_0 = \sum_{j} \alpha'_j$:
\begin{align*}
    \alpha_0 = \alpha'_A + \alpha'_B + \alpha'_C = 3 + 3 + 1 = 7
\end{align*}
Now, we can find the expected value for each candidate's vote share:
\begin{align*}
    \mathbb{E}[\theta_A \mid X] &= \frac{\alpha'_A}{\alpha_0} = \frac{3}{7} \\
    \mathbb{E}[\theta_B \mid X] &= \frac{\alpha'_B}{\alpha_0} = \frac{3}{7} \\
    \mathbb{E}[\theta_C \mid X] &= \frac{\alpha'_C}{\alpha_0} = \frac{1}{7}
\end{align*}
The expected value of the vector $\left(\theta_{A}, \theta_{B}, \theta_{C}\right)$ is the vector of these expected values.

\subsubsection*{\normalfont}{$\therefore$ The expected value is $\mathbb{E}[\left(\theta_{A}, \theta_{B}, \theta_{C}\right) \mid X] = \left(\frac{3}{7}, \frac{3}{7}, \frac{1}{7}\right)$.} \\

\subsubsection*{6 (a): $\ell_1$ for $p$ and $q$}

\parbox{\textwidth}{The $\ell_1$ distance between two vectors $p, q \in \mathbb{R}^n$ is given by:
$$\|p-q\|_1 = \sum_{i=1}^n |p_i - q_i|$$

Let $p = \begin{bmatrix} 1/2 \\ 1/4 \\ 1/8 \\ 1/8 \end{bmatrix}$ and $q = \begin{bmatrix} 1/4 \\ 1/4 \\ 1/4 \\ 1/4 \end{bmatrix}$

\begin{align*}
    \|p-q\|_1 &= \left|\frac{1}{2} - \frac{1}{4}\right| + \left|\frac{1}{4} - \frac{1}{4}\right| + \left|\frac{1}{8} - \frac{1}{4}\right| + \left|\frac{1}{8} - \frac{1}{4}\right| \\
    &= \left|\frac{2}{4} - \frac{1}{4}\right| + |0| + \left|\frac{1}{8} - \frac{2}{8}\right| + \left|\frac{1}{8} - \frac{2}{8}\right| \\
    &= \frac{1}{4} + 0 + \left|-\frac{1}{8}\right| + \left|-\frac{1}{8}\right| \\
    &= \frac{1}{4} + \frac{1}{8} + \frac{1}{8} = \frac{2}{8} + \frac{1}{8} + \frac{1}{8} = \frac{4}{8} = \frac{1}{2}
\end{align*}
}
\subsubsection*{\normalfont}{$\therefore$ $\ell_{1}= \frac{1}{2}$}

\parbox{\textwidth}{The $\ell_1$ distance between two vectors $q, r \in \mathbb{R}^n$ is given by:
$$\|q-r\|_1 = \sum_{i=1}^n |q_i - r_i|$$

Let $q = \begin{bmatrix} 1/4 \\ 1/4 \\ 1/4 \\ 1/4 \end{bmatrix}$ and $r = \begin{bmatrix} 1/2 \\ 0 \\ 1/4 \\ 1/4 \end{bmatrix}$

\subsubsection*{6 (c): KL divergence $K(p, q)$}
\parbox{\textwidth}{
The Kullback-Leibler (KL) divergence from a distribution $p$ to a distribution $q$ is defined as:
$$K(p, q) = \sum_{i} p_i \ln \frac{p_i}{q_i}$$
Let $p = \begin{bmatrix} 1/2 \\ 1/4 \\ 1/8 \\ 1/8 \end{bmatrix}$ and $q = \begin{bmatrix} 1/4 \\ 1/4 \\ 1/4 \\ 1/4 \end{bmatrix}$
}

\begin{align*}
K(p, q) &= \sum_{i=1}^{4} p_i \ln\left(\frac{p_i}{q_i}\right) \\
&= p_1 \ln\left(\frac{p_1}{q_1}\right) + p_2 \ln\left(\frac{p_2}{q_2}\right) + p_3 \ln\left(\frac{p_3}{q_3}\right) + p_4 \ln\left(\frac{p_4}{q_4}\right) \\
&= \frac{1}{2}\ln\left(\frac{1/2}{1/4}\right) + \frac{1}{4}\ln\left(\frac{1/4}{1/4}\right) + \frac{1}{8}\ln\left(\frac{1/8}{1/4}\right) + \frac{1}{8}\ln\left(\frac{1/8}{1/4}\right) \\
&= \frac{1}{2}\ln(2) + \frac{1}{4}\ln(1) + \frac{1}{8}\ln\left(\frac{1}{2}\right) + \frac{1}{8}\ln\left(\frac{1}{2}\right) \\
&= \frac{1}{2}\ln(2) + \frac{1}{4}(0) - \frac{1}{8}\ln(2) - \frac{1}{8}\ln(2) \\
&= \frac{1}{2}\ln(2) - \frac{2}{8}\ln(2) \\
&= \left(\frac{1}{2} - \frac{1}{4}\right)\ln(2) \\
&= \frac{1}{4}\ln(2)
\end{align*}

\subsubsection*{\normalfont{$\therefore$ $K(p,q) = \frac{1}{4}\ln(2)$}}

\parbox{\textwidth}{
The Kullback-Leibler (KL) divergence from a distribution $q$ to a distribution $r$ is defined as:
$$K(q, r) = \sum_{i} q_i \ln \frac{q_i}{r_i}$$
Let $q = \begin{bmatrix} 1/4 \\ 1/4 \\ 1/4 \\ 1/4 \end{bmatrix}$ and $r = \begin{bmatrix} 1/2 \\ 0 \\ 1/4 \\ 1/4 \end{bmatrix}$ \\

Hence, the divergence will be infinite.
}
\subsubsection*{\normalfont}{$\therefore$ $K(q, r) = \infty$}
